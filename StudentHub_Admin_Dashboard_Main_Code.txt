// php/auth.php

function requireAdmin(): array
{
    if (!isAuthenticated()) {
        studenthub_require_authentication("admin", 401);
    }

    if ($_SESSION["user_type"] !== "admin") {
        studenthub_require_authentication("admin", 403);
    }

    $connection = studenthub_db("studenthub");
    $statement = $connection->prepare(
        "SELECT id, full_name, username, email, role
         FROM admins
         WHERE id = ?
           AND status = 'active'
           AND role IN ('admin', 'super_admin')
         LIMIT 1"
    );

    $statement->bind_param("i", $_SESSION["user_id"]);
    $statement->execute();
    $admin = $statement->get_result()->fetch_assoc();
    $statement->close();

    if (!$admin || $admin["role"] !== $_SESSION["role"]) {
        expireCurrentSession();
        studenthub_require_authentication("admin", 401);
    }

    return $admin;
}

4. LATEST REGISTRATIONS
-----------------------

$recentRegistrations = $connection->query(
    "SELECT r.student_id,
            s.student_name,
            s.email,
            r.event_title,
            r.event_date,
            r.time_stamp AS registration_date
     FROM event_registrations r
     LEFT JOIN student s ON s.student_id = r.student_id
     ORDER BY r.time_stamp DESC, r.registration_id DESC
     LIMIT 6"
)->fetch_all(MYSQLI_ASSOC);


5. REGISTRATION ACTIVITY FOR DASHBOARD GRAPH
---------------------------------------------

$monthlyRows = $connection->query(
    "SELECT DATE_FORMAT(time_stamp, '%Y-%m') AS month_key,
            COUNT(*) AS registration_count
     FROM event_registrations
     WHERE time_stamp >= DATE_FORMAT(
         DATE_SUB(CURRENT_DATE, INTERVAL 5 MONTH), '%Y-%m-01'
     )
     GROUP BY DATE_FORMAT(time_stamp, '%Y-%m')
     ORDER BY month_key"
)->fetch_all(MYSQLI_ASSOC);

$monthlyCounts = [];
foreach ($monthlyRows as $row) {
    $monthlyCounts[$row["month_key"]] =
        (int) $row["registration_count"];
}

$registrationActivity = [];
$monthStart = new DateTimeImmutable("first day of this month");

for ($monthOffset = 5; $monthOffset >= 0; $monthOffset--) {
    $month = $monthStart->modify("-" . $monthOffset . " months");
    $monthKey = $month->format("Y-m");

    $registrationActivity[] = [
        "label" => $month->format("M Y"),
        "count" => $monthlyCounts[$monthKey] ?? 0,
    ];
}
