{

// html/admin/includes/bootstrap.php
function adminEventInput(): array
{
    $read = static function (string $key): string {
        $value = $_POST[$key] ?? "";
        return is_string($value) ? trim($value) : "";
    };

    return [
        "title" => $read("title"),
        "event_date" => $read("event_date"),
        "event_time" => $read("event_time"),
        "venue" => $read("venue"),
        "category" => $read("category"),
        "organizer" => $read("organizer"),
        "description" => $read("description"),
    ];
}

function adminEventErrors(array $event): array
{
    $errors = [];
    $date = DateTime::createFromFormat("Y-m-d", $event["event_date"]);
    $dateErrors = DateTime::getLastErrors();

    if ($event["title"] === "" || strlen($event["title"]) > 150) {
        $errors[] = "Title is required and must be 150 characters or fewer.";
    }

    if (
        !$date
        || ($dateErrors !== false && (
            $dateErrors["warning_count"] > 0
            || $dateErrors["error_count"] > 0
        ))
        || ($date && $date->format("Y-m-d") !== $event["event_date"])
    ) {
        $errors[] = "Enter a valid event date.";
    }

    if ($event["event_time"] === "" || strlen($event["event_time"]) > 20) {
        $errors[] = "Enter a valid event time.";
    }

    if ($event["venue"] === "" || strlen($event["venue"]) > 150) {
        $errors[] = "Venue is required and must be 150 characters or fewer.";
    }

    if ($event["category"] === "" || strlen($event["category"]) > 50) {
        $errors[] = "Category is required and must be 50 characters or fewer.";
    }

    if (strlen($event["organizer"]) > 150) {
        $errors[] = "Organizer must be 150 characters or fewer.";
    }

    if ($event["description"] === "") {
        $errors[] = "Description is required.";
    }

    return $errors;
}

// html/admin/includes/bootstrap.php
function adminStoreEventPoster(array $upload): ?string
{
    if (($upload["error"] ?? UPLOAD_ERR_NO_FILE) === UPLOAD_ERR_NO_FILE) {
        return null;
    }

    if (
        ($upload["error"] ?? UPLOAD_ERR_NO_FILE) !== UPLOAD_ERR_OK
        || ($upload["size"] ?? 0) > 5 * 1024 * 1024
    ) {
        throw new RuntimeException("Choose an image smaller than 5 MB.");
    }

    $fileInfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $fileInfo->file($upload["tmp_name"]);
    $extensions = [
        "image/jpeg" => "jpg",
        "image/png" => "png",
        "image/webp" => "webp",
    ];

    if (!isset($extensions[$mimeType])) {
        throw new RuntimeException(
            "Poster must be a JPEG, PNG, or WebP image."
        );
    }

    $fileName = bin2hex(random_bytes(16))
        . "." . $extensions[$mimeType];
    $directory = dirname(__DIR__, 2) . "/uploads/events";

    if (
        !is_dir($directory)
        && !mkdir($directory, 0755, true)
        && !is_dir($directory)
    ) {
        throw new RuntimeException("Poster storage is unavailable.");
    }

    if (
        !move_uploaded_file(
            $upload["tmp_name"],
            $directory . "/" . $fileName
        )
    ) {
        throw new RuntimeException("Poster could not be saved.");
    }

    return "uploads/events/" . $fileName;
}

// html/admin/event-create.php
if ($_SERVER["REQUEST_METHOD"] === "POST") {
    adminRequireCsrf();

    $eventData = adminEventInput();
    $errors = adminEventErrors($eventData);
    $posterPath = null;

    if (!$errors) {
        try {
            $posterPath = adminStoreEventPoster($_FILES["poster"] ?? []);
        } catch (RuntimeException $exception) {
            $errors[] = $exception->getMessage();
        }
    }

    if (!$errors) {
        try {
            $statement = $connection->prepare(
                "INSERT INTO events
                 (title, event_date, event_time, venue, category,
                  organizer, description, image)
                 VALUES (?, ?, ?, ?, ?, ?, ?, ?)"
            );

            $statement->bind_param(
                "ssssssss",
                $eventData["title"],
                $eventData["event_date"],
                $eventData["event_time"],
                $eventData["venue"],
                $eventData["category"],
                $eventData["organizer"],
                $eventData["description"],
                $posterPath
            );

            $statement->execute();
            $eventId = (int) $connection->insert_id;
            $statement->close();

            adminFlash("Event created successfully.");
            adminRedirect("event-view.php?id=" . $eventId);
        } catch (mysqli_sql_exception $exception) {
            error_log("Event creation failed: " . $exception->getMessage());
            $errors[] = "Event could not be created.";
        }
    }
}

// html/admin/event-view.php
$eventId = filter_var(
    $_GET["id"] ?? null,
    FILTER_VALIDATE_INT
);

$statement = $connection->prepare(
    "SELECT event_id, title, event_date, event_time, venue, category,
            organizer, description, image
     FROM events
     WHERE event_id = ?
     LIMIT 1"
);

$statement->bind_param("i", $eventId);
$statement->execute();
$event = $statement->get_result()->fetch_assoc();
$statement->close();

if (!$event) {
    http_response_code(404);
    exit("Event not found.");
}

// Main displayed values:
echo adminEscape($event["title"]);
echo adminEscape($event["event_date"]);
echo adminEscape($event["event_time"]);
echo adminEscape($event["venue"]);
echo adminEscape($event["category"]);

    echo '<img src="../' . adminEscape($event["image"])
        . '" alt="' . adminEscape($event["title"]) . '">';

// html/admin/event-edit.php
$eventId = filter_var(
    $_GET["id"] ?? $_POST["event_id"] ?? null,
    FILTER_VALIDATE_INT
);

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    adminRequireCsrf();

    $eventData = array_merge(
        $eventData,
        adminEventInput()
    );
    $errors = adminEventErrors($eventData);
    $newPoster = null;

    if (!$errors) {
        try {
            $newPoster = adminStoreEventPoster($_FILES["poster"] ?? []);
        } catch (RuntimeException $exception) {
            $errors[] = $exception->getMessage();
        }
    }

    if (!$errors) {
        $image = $newPoster ?? $eventData["image"];
        $update = $connection->prepare(
            "UPDATE events
             SET title = ?, event_date = ?, event_time = ?, venue = ?,
                 category = ?, organizer = ?, description = ?, image = ?
             WHERE event_id = ?"
        );

        $update->bind_param(
            "ssssssssi",
            $eventData["title"],
            $eventData["event_date"],
            $eventData["event_time"],
            $eventData["venue"],
            $eventData["category"],
            $eventData["organizer"],
            $eventData["description"],
            $image,
            $eventId
        );

        $update->execute();
        $update->close();

        adminFlash("Event updated successfully.");
        adminRedirect("event-view.php?id=" . $eventId);
    }
}

// html/admin/event-delete.php
adminRequirePost();
adminRequireCsrf();

$eventId = filter_var(
    $_POST["event_id"] ?? null,
    FILTER_VALIDATE_INT
);

$posterQuery = $connection->prepare(
    "SELECT image FROM events
     WHERE event_id = ?
     LIMIT 1"
);
$posterQuery->bind_param("i", $eventId);
$posterQuery->execute();
$poster = $posterQuery->get_result()->fetch_column();
$posterQuery->close();

$statement = $connection->prepare(
    "DELETE FROM events WHERE event_id = ?"
);
$statement->bind_param("i", $eventId);
$statement->execute();
$deleted = $statement->affected_rows > 0;
$statement->close();

if ($deleted) {
    if (
        is_string($poster)
        && str_starts_with($poster, "uploads/events/")
    ) {
        $localPoster = dirname(__DIR__) . "/" . $poster;
        if (is_file($localPoster)) {
            unlink($localPoster);
        }
    }
    adminFlash("Event deleted successfully.");
} else {
    adminFlash("Event not found.", "error");
}

adminRedirect("events.php");
