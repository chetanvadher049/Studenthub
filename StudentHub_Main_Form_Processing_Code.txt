STUDENTHUB - MAIN PHP FORM PROCESSING CODE
============================================

Aim:
Process registration/contact form data using PHP, validate and sanitize inputs on
server side, store records, and display success/error messages.

Note: In the existing StudentHub project, records are stored in MySQL using
prepared statements. The project does not currently use CSV or JSON storage.


1. REGISTRATION FORM PROCESSING (php/register.php)
--------------------------------------------------

<?php
require_once __DIR__ . "/auth.php";
studenthub_start_session();

try {
    // Accept only submitted form data.
    if ($_SERVER["REQUEST_METHOD"] !== "POST") {
        http_response_code(405);
        header("Allow: POST");
        registrationError(
            "Invalid request",
            "Submit the registration form to create your account.",
            405
        );
    }

    // Verify the CSRF token before processing the form.
    if (!studenthub_verify_csrf_token($_POST["csrf_token"] ?? null)) {
        registrationError(
            "Registration form expired",
            "Reload the registration page and submit the form again.",
            403
        );
    }

    // Read and sanitize submitted values.
    $studentName = trim(postedString("student_name"));
    $username = trim(postedString("username"));
    $email = trim(postedString("email"));
    $mobileNumber = trim(postedString("mobile_no"));
    $gender = trim(postedString("gender"));
    $course = trim(postedString("course"));
    $academicYear = trim(postedString("academic_year"));
    $password = postedString("password");
    $confirmPassword = postedString("confirm_password");
    $termsAccepted = isset($_POST["terms"]);

    // Server-side validation.
    if (!preg_match(
        "/^[\\p{L}\\p{M}][\\p{L}\\p{M}\\s.'?\\-]{1,99}$/u",
        $studentName
    )) {
        registrationError(
            "Check your name",
            "Enter a valid name between 2 and 100 characters."
        );
    }

    if (!preg_match("/^[A-Za-z0-9_]{3,30}$/", $username)) {
        registrationError(
            "Check your username",
            "Use 3 to 30 letters, numbers or underscores."
        );
    }

    if (!filter_var($email, FILTER_VALIDATE_EMAIL) || strlen($email) > 100) {
        registrationError(
            "Check your email",
            "Enter a valid email address of no more than 100 characters."
        );
    }

    if (!preg_match("/^[0-9]{10}$/", $mobileNumber)) {
        registrationError(
            "Check your mobile number",
            "Enter exactly 10 digits."
        );
    }

    $allowedGender = ["Male", "Female", "Other"];
    if (!in_array($gender, $allowedGender, true)) {
        registrationError("Choose a gender", "Select a valid gender.");
    }

    if ($course === "" || strlen($course) > 100) {
        registrationError("Check your course", "Enter a valid course name.");
    }

    if ($academicYear === "" || strlen($academicYear) > 30) {
        registrationError("Check your academic year", "Enter a valid academic year.");
    }

    if (
        strlen($password) < 8 ||
        strlen($password) > 100 ||
        !preg_match("/[a-z]/", $password) ||
        !preg_match("/[A-Z]/", $password) ||
        !preg_match("/[0-9]/", $password) ||
        !preg_match("/[@$!%*?&]/", $password)
    ) {
        registrationError(
            "Choose a stronger password",
            "Use 8 to 100 characters with uppercase, lowercase, a number and a special character."
        );
    }

    if ($password !== $confirmPassword) {
        registrationError(
            "Passwords do not match",
            "Enter the same password in both password fields."
        );
    }

    if (!$termsAccepted) {
        registrationError(
            "Terms required",
            "Accept the Terms and Conditions before registering."
        );
    }

    // Store the record safely in the database.
    $conn = studenthub_db();
    $passwordHash = password_hash($password, PASSWORD_DEFAULT);

    $statement = $conn->prepare(
        "INSERT INTO students
        (full_name, username, email, password_hash, phone, gender, course, academic_year)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)"
    );

    $statement->bind_param(
        "ssssssss",
        $studentName,
        $username,
        $email,
        $passwordHash,
        $mobileNumber,
        $gender,
        $course,
        $academicYear
    );

    $statement->execute();
    $statement->close();
    $conn->close();

    // Display success result through the login page.
    header("Location: ../html/login.html?registered=1");
    exit;

} catch (mysqli_sql_exception $exception) {
    error_log("StudentHub registration database error: " . $exception->getMessage());
    registrationError(
        "Registration unavailable",
        "We could not create your account right now. Please try again later.",
        500
    );
}

function postedString(string $key): string
{
    $value = $_POST[$key] ?? "";
    return is_string($value) ? $value : "";
}

function registrationError(
    string $title,
    string $message,
    int $statusCode = 400
): void {
    http_response_code($statusCode);
    $safeTitle = htmlspecialchars($title, ENT_QUOTES, "UTF-8");
    $safeMessage = htmlspecialchars($message, ENT_QUOTES, "UTF-8");
    ?>
    <!DOCTYPE html>
    <html lang="en">
    <head><meta charset="UTF-8"><title><?= $safeTitle ?></title></head>
    <body>
        <main>
            <h1 role="alert"><?= $safeTitle ?></h1>
            <p><?= $safeMessage ?></p>
            <a href="../html/register.html">Back to registration</a>
        </main>
    </body>
    </html>
    <?php
    exit;
}
?>


2. CONTACT FORM PROCESSING (php/contact.php)
---------------------------------------------

<?php
require_once __DIR__ . "/auth.php";
studenthub_start_session();

if ($_SERVER["REQUEST_METHOD"] !== "POST") {
    http_response_code(405);
    header("Allow: POST");
    contactResult("Invalid request", "Submit the contact form.", false, 405);
}

if (!verifyCsrfToken($_POST["csrf_token"] ?? null)) {
    contactResult(
        "Form expired",
        "Reload the contact page and try again.",
        false,
        403
    );
}

// Read and sanitize input values.
$name = contactInput("name");
$email = trim(contactInput("email"));
$subject = contactInput("subject");
$message = contactInput("message");
$honeypot = contactInput("company");

// Reject automated submissions.
if ($honeypot !== "") {
    contactResult(
        "Message not accepted",
        "Please submit the contact form directly.",
        false,
        400
    );
}

// Validate the submitted values.
if (
    !preg_match("/^[\\p{L}\\p{M}][\\p{L}\\p{M}\\s.'?\\-]{1,99}$/u", $name) ||
    !filter_var($email, FILTER_VALIDATE_EMAIL) ||
    strlen($email) > 100 ||
    $subject === "" ||
    strlen($subject) > 150 ||
    trim($message) === "" ||
    strlen($message) > 20000
) {
    contactResult(
        "Check your message",
        "Enter a valid name, email, subject and message.",
        false,
        400
    );
}

try {
    // Store the contact message using a prepared statement.
    $connection = studenthub_db();
    $statement = $connection->prepare(
        "INSERT INTO contact_messages (name, email, subject, message)
         VALUES (?, ?, ?, ?)"
    );
    $statement->bind_param("ssss", $name, $email, $subject, $message);
    $statement->execute();
    $statement->close();
    $connection->close();
} catch (mysqli_sql_exception $exception) {
    error_log("StudentHub contact submission failed: " . $exception->getMessage());
    contactResult(
        "Message could not be sent",
        "We could not save your message right now. Please try again later.",
        false,
        500
    );
}

// Display success message.
contactResult(
    "Message received",
    "Thank you for contacting StudentHub. Your message has been saved.",
    true
);

function contactInput(string $key): string
{
    $value = $_POST[$key] ?? "";
    return is_string($value) ? trim($value) : "";
}

function contactResult(
    string $title,
    string $message,
    bool $success,
    int $statusCode = 200
): void {
    http_response_code($statusCode);
    $safeTitle = htmlspecialchars($title, ENT_QUOTES, "UTF-8");
    $safeMessage = htmlspecialchars($message, ENT_QUOTES, "UTF-8");
    ?>
    <!DOCTYPE html>
    <html lang="en">
    <head><meta charset="UTF-8"><title><?= $safeTitle ?></title></head>
    <body>
        <main>
            <h1 role="<?= $success ? 'status' : 'alert' ?>">
                <?= $safeTitle ?>
            </h1>
            <p><?= $safeMessage ?></p>
            <a href="../html/contact.html">Back to contact</a>
        </main>
    </body>
    </html>
    <?php
    exit;
}
?>
