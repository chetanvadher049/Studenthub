<?php
require_once __DIR__ . "/auth.php";
studenthub_start_session();

try {
    if ($_SERVER["REQUEST_METHOD"] !== "POST") {
        http_response_code(405);
        header("Allow: POST");
        registrationError(

    }

    if (!studenthub_verify_csrf_token($_POST["csrf_token"] ?? null)) {
        registrationError(
            "Registration form expired",
            "Reload the registration page and submit the form again.",
            403
        );
    }

    $studentName = trim(postedString("student_name"));
    $username = trim(postedString("username"));
    $email = trim(postedString("email"));
    $mobileNumber = trim(postedString("mobile_no"));
    $gender = trim(postedString("gender"));
    $course = trim(postedString("course"));
    $academicYear = trim(postedString("academic_year"));
    $password = postedString("password");
    $confirmPassword = postedString("confirm_password");
    $termsAccepted = isset($_POST["terms"]);

    if (
        !preg_match(
            "/^[\\p{L}\\p{M}][\\p{L}\\p{M}\\s.'’\\-]{1,99}$/u",
            $studentName
        )
    ) {
        registrationError(
            "Check your name",
            "Enter a name between 2 and 100 characters using letters, spaces, apostrophes or hyphens."
        );
    }

    if (!preg_match("/^[A-Za-z0-9_]{3,30}$/", $username)) {
        registrationError(
            "Check your username",
            "Use 3 to 30 letters, numbers or underscores for your username."
        );
    }

    if (!filter_var($email, FILTER_VALIDATE_EMAIL) || strlen($email) > 100) {
        registrationError(
            "Check your email",
            "Enter a valid email address of no more than 100 characters."
        );
    }

    if (!preg_match("/^[0-9]{10}$/", $mobileNumber)) {
        registrationError(
            "Check your mobile number",
            "Enter exactly 10 digits."
        );
    }

    $allowed_gender = ["Male", "Female", "Other"];

    if (!in_array($gender, $allowed_gender, true)) {
        registrationError("Choose a gender", "Select one of the available options.");
    }

    if ($course === "" || strlen($course) > 100) {
        registrationError(
            "Check your course",
            "Enter a course name of no more than 100 characters."
        );
    }

    if ($academicYear === "" || strlen($academicYear) > 30) {
        registrationError(
            "Check your academic year",
            "Enter an academic year of no more than 30 characters."
        );
    }


    if (
        strlen($password) < 8 ||
        strlen($password) > 100 ||
        !preg_match("/[a-z]/", $password) ||
        !preg_match("/[A-Z]/", $password) ||
        !preg_match("/[0-9]/", $password) ||
        !preg_match("/[@$!%*?&]/", $password)
    ) {
        registrationError(
            "Choose a stronger password",
            "Use 8 to 100 characters, including uppercase and lowercase letters, a number, and one of @$!%*?&."
        );
    }


    if ($password !== $confirmPassword) {
        registrationError(
            "Passwords do not match",
            "Enter the same password in both password fields."
        );
    }

    if (!$termsAccepted) {
        registrationError(
            "Terms required",
            "Accept the Terms & Conditions before creating your account."
        );
    }

    $conn = studenthub_db();

    $existing_account = $conn->prepare(
        "SELECT email, username
         FROM students
         WHERE email = ? OR username = ?
         LIMIT 1"
    );

    $existing_account->bind_param("ss", $email, $username);
    $existing_account->execute();
    $existing_account->store_result();

    if ($existing_account->num_rows > 0) {
        $existing_account->close();
        $conn->close();

        registrationError(
            "Account details already in use",
            "An account already uses this email address or username. Choose different details or sign in.",
            409
        );
    }

    $existing_account->close();

    $passwordHash = password_hash($password, PASSWORD_DEFAULT);

    $stmt = $conn->prepare(
        "INSERT INTO students
        (
            full_name,
            username,
            email,
            password_hash,
            phone,
            gender,
            course,
            academic_year
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)"
    );

    $stmt->bind_param(
        "ssssssss",
        $studentName,
        $username,
        $email,
        $passwordHash,
        $mobileNumber,
        $gender,
        $course,
        $academicYear
    );

    $stmt->execute();

    $stmt->close();
    $conn->close();

    header("Location: ../html/login.html?registered=1");
    exit;

if (
    !preg_match("/^[\\p{L}\\p{M}][\\p{L}\\p{M}\\s.'’\\-]{1,99}$/u", $name) ||
    !filter_var($email, FILTER_VALIDATE_EMAIL) ||
    strlen($email) > 100 ||
    $subject === "" ||

    trim($message) === "" ||

) {
    contactResult(
        "Check your message",
        "Enter a valid name and email, a subject of up to 150 characters, and a message of up to 5,000 characters.",
        false,
        400
    );
}

try {
    $connection = studenthub_db();
    $statement = $connection->prepare(
        "INSERT INTO contact_messages (name, email, subject, message)
         VALUES (?, ?, ?, ?)"

        "Message could not be sent",

    "Message received",

    true
);

3. CSRF and session helpers - selected code from php/auth.php
=============================================================

function studenthub_csrf_token(): string
{
    studenthub_start_session();

    if (!is_string($_SESSION["studenthub_csrf"] ?? null)) {
        $_SESSION["studenthub_csrf"] = bin2hex(random_bytes(32));
    }

    return $_SESSION["studenthub_csrf"];
}

function csrfToken(): string
{
    return studenthub_csrf_token();
}

function verifyCsrfToken($token): bool
{
    return is_string($token)
        && isset($_SESSION["studenthub_csrf"])
        && is_string($_SESSION["studenthub_csrf"])
        && hash_equals($_SESSION["studenthub_csrf"], $token);
}

function studenthub_verify_csrf_token($token): bool
{
    return verifyCsrfToken($token);
}

4. Database connection - php/dbconnect.php
==========================================

<?php

function studenthub_db(?string $databaseOverride = null): mysqli
{
    static $connections = [];

    mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT);

    $host = getenv("STUDENTHUB_DB_HOST") ?: "127.0.0.1";
    $username = getenv("STUDENTHUB_DB_USER") ?: "root";
    $password = getenv("STUDENTHUB_DB_PASSWORD") ?: "";
    $database = $databaseOverride
        ?? (getenv("STUDENTHUB_DB_NAME") ?: "studenthub_db");
    $port = (int) (getenv("STUDENTHUB_DB_PORT") ?: 3306);
    $connectionKey = $host . ":" . $port . "/" . $database . "/" . $username;

    if (isset($connections[$connectionKey])) {
        try {
            $connections[$connectionKey]->query("SELECT 1");
            return $connections[$connectionKey];
        } catch (mysqli_sql_exception $exception) {
            if (!in_array((int) $exception->getCode(), [2006, 2013, 2055], true)) {
                throw $exception;
            }
            unset($connections[$connectionKey]);
        }
    }

    try {
        $connection = new mysqli(
            $host,
            $username,
            $password,
            $database,
            $port
        );
    } catch (mysqli_sql_exception $exception) {
        error_log("StudentHub database connection failed: " . $exception->getMessage());
        throw new mysqli_sql_exception("StudentHub database connection failed.");
    }

    $connection->set_charset("utf8mb4");
    $connections[$connectionKey] = $connection;

    return $connection;
}
