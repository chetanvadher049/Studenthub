// php/dbconnect.php
function studenthub_db(?string $databaseOverride = null): mysqli
{
    mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT);

    $host = getenv("STUDENTHUB_DB_HOST") ?: "127.0.0.1";
    $username = getenv("STUDENTHUB_DB_USER") ?: "root";
    $password = getenv("STUDENTHUB_DB_PASSWORD") ?: "";
    $database = $databaseOverride
        ?? (getenv("STUDENTHUB_DB_NAME") ?: "studenthub");
    $port = (int) (getenv("STUDENTHUB_DB_PORT") ?: 3306);

    $connection = new mysqli(
        $host,
        $username,
        $password,
        $database,
        $port
    );
    $connection->set_charset("utf8mb4");

    return $connection;
}

function adminEscape($value): string
{
    return htmlspecialchars((string) $value, ENT_QUOTES, "UTF-8");
}

function adminStudentInput(): array
{
    $read = static function (string $key): string {
        $value = $_POST[$key] ?? "";
        return is_string($value) ? trim($value) : "";
    };

    return [
        "full_name" => $read("full_name"),
        "username" => $read("username"),
        "email" => $read("email"),
        "phone" => $read("phone"),
        "course" => $read("course"),
        "gender" => $read("gender"),
        "academic_year" => $read("academic_year"),
        "status" => $read("status"),
        "password" => is_string($_POST["password"] ?? null)
            ? $_POST["password"] : "",
    ];
}

function adminStudentErrors(array $student, bool $passwordRequired): array
{
    $errors = [];

    if (!preg_match(
        "/^[\\p{L}\\p{M}][\\p{L}\\p{M}\\s.'’\\-]{1,99}$/u",
        $student["full_name"]
    )) {
        $errors[] = "Enter a valid full name between 2 and 100 characters.";
    }

    if (!preg_match(
        "/^[A-Za-z0-9_]{3,50}$/",
        $student["username"]
    )) {
        $errors[] = "Use 3 to 50 letters, numbers, or underscores.";
    }

    if (
        !filter_var($student["email"], FILTER_VALIDATE_EMAIL)
        || strlen($student["email"]) > 255
    ) {
        $errors[] = "Enter a valid email address.";
    }

    if (
        $student["phone"] !== ""
        && !preg_match("/^[0-9]{10}$/", $student["phone"])
    ) {
        $errors[] = "Enter a valid 10-digit mobile number.";
    }

    if (
        $passwordRequired
        || $student["password"] !== ""
    ) {
        if (
            strlen($student["password"]) < 8
            || !preg_match("/[a-z]/", $student["password"])
            || !preg_match("/[A-Z]/", $student["password"])
            || !preg_match("/[0-9]/", $student["password"])
            || !preg_match("/[@$!%*?&]/", $student["password"])
        ) {
            $errors[] = "Password does not meet the required format.";
        }
    }

    return $errors;
}

function adminFlash(string $message, string $type = "success"): void
{
    $_SESSION["admin_flash"] = [
        "message" => $message,
        "type" => $type
    ];
}

// html/admin/student-create.php
if ($_SERVER["REQUEST_METHOD"] === "POST") {
    adminRequireCsrf();

    $studentData = adminStudentInput();
    $errors = adminStudentErrors($studentData, true);

    if (!$errors) {
        try {
            $studentId = "STU_" . bin2hex(random_bytes(12));
            $passwordHash = password_hash(
                $studentData["password"],
                PASSWORD_DEFAULT
            );

            $statement = $connection->prepare(
                "INSERT INTO student
                (student_id, student_name, username, email, mobile_no,
                 gender, course, academic_year, password)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)"
            );

            $statement->bind_param(
                "sssssssss",
                $studentId,
                $studentData["full_name"],
                $studentData["username"],
                $studentData["email"],
                $studentData["phone"],
                $studentData["gender"],
                $studentData["course"],
                $studentData["academic_year"],
                $passwordHash
            );

            $statement->execute();
            $statement->close();

            adminFlash("Student created successfully.");
            adminRedirect(
                "student-view.php?id=" . urlencode($studentId)
            );
        } catch (mysqli_sql_exception $exception) {
            $errors[] = (int) $exception->getCode() === 1062
                ? "Username or email is already in use."
                : "Student could not be created.";
            error_log(
                "Student creation failed: " . $exception->getMessage()
            );
        }
    }
}

// html/admin/student-edit.php
$studentId = trim((string) ($_GET["id"] ?? $_POST["student_id"] ?? ""));

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    adminRequireCsrf();

    $input = adminStudentInput();
    $errors = adminStudentErrors($input, false);

    if (!$errors) {
        $check = $connection->prepare(
            "SELECT student_id
             FROM student
             WHERE (email = ? OR username = ?)
               AND student_id <> ?
             LIMIT 1"
        );
        $check->bind_param(
            "sss",
            $input["email"],
            $input["username"],
            $studentId
        );
        $check->execute();
        $check->store_result();

        if ($check->num_rows > 0) {
            $errors[] = "Username or email is already in use.";
        }
        $check->close();
    }

    if (!$errors) {
        if ($input["password"] === "") {
            $update = $connection->prepare(
                "UPDATE student
                 SET student_name = ?, username = ?, email = ?,
                     mobile_no = ?, gender = ?, course = ?,
                     academic_year = ?
                 WHERE student_id = ?"
            );
            $update->bind_param(
                "ssssssss",
                $input["full_name"],
                $input["username"],
                $input["email"],
                $input["phone"],
                $input["gender"],
                $input["course"],
                $input["academic_year"],
                $studentId
            );
        } else {
            $passwordHash = password_hash(
                $input["password"],
                PASSWORD_DEFAULT
            );
            $update = $connection->prepare(
                "UPDATE student
                 SET student_name = ?, username = ?, email = ?,
                     mobile_no = ?, gender = ?, course = ?,
                     academic_year = ?, password = ?
                 WHERE student_id = ?"
            );
            $update->bind_param(
                "sssssssss",
                $input["full_name"],
                $input["username"],
                $input["email"],
                $input["phone"],
                $input["gender"],
                $input["course"],
                $input["academic_year"],
                $passwordHash,
                $studentId
            );
        }

        $update->execute();
        $update->close();

        adminFlash("Student updated successfully.");
        adminRedirect(
            "student-view.php?id=" . urlencode($studentId)
        );
    }
}

// html/admin/students.php
$search = trim((string) ($_GET["q"] ?? ""));
$courseFilter = trim((string) ($_GET["course"] ?? ""));

$conditions = [];
$parameters = [];
$types = "";

if ($search !== "") {
    $conditions[] = "
        (student_id LIKE ?
         OR student_name LIKE ?
         OR username LIKE ?
         OR email LIKE ?
         OR mobile_no LIKE ?)
    ";
    $term = "%{$search}%";
    $parameters = [$term, $term, $term, $term, $term];
    $types = "sssss";
}

if ($courseFilter !== "") {
    $conditions[] = "course = ?";
    $parameters[] = $courseFilter;
    $types .= "s";
}

$where = $conditions
    ? " WHERE " . implode(" AND ", $conditions)
    : "";

$statement = $connection->prepare(
    "SELECT student_id, student_name, username, email, mobile_no,
            gender, course, academic_year, time_stamp
     FROM student"
    . $where
    . " ORDER BY time_stamp DESC"
);

if ($types !== "") {
    $statement->bind_param($types, ...$parameters);
}

$statement->execute();
$students = $statement->get_result()->fetch_all(MYSQLI_ASSOC);
$statement->close();

// Populate the course filter using MySQLi:
$courseResult = $connection->query(
    "SELECT DISTINCT course
     FROM student
     WHERE course <> ''
     ORDER BY course"
);
$courses = array_column(
    $courseResult->fetch_all(MYSQLI_NUM),
    0
);
$courseResult->free();
